# Step 1: Check node version
node --version

# Step 2: Set execution policy (Windows only)
Set-ExecutionPolicy RemoteSigned -Scope CurrentUser

# Step 3: Create the expo app
pnpm create expo-app --template blank

# Step 4: Move into the project
Set-Location .\my-app

# Step 5: Install all npm packages
pnpm add dotenv @react-navigation/native @react-navigation/stack @react-navigation/bottom-tabs @expo/vector-icons @react-native-community/slider

# Step 6: Install all expo packages
pnpm dlx expo install react-native-gesture-handler @react-native-masked-view/masked-view react-native-screens react-native-safe-area-context react-dom react-native-web expo-splash-screen expo-font expo-status-bar

# Step 7: Start the app
pnpm dlx expo start
